{{- /* resume, sinon premier paragraphe non vide, sans Markdown ni HTML, coupé à ~160 caractères à la fin d'un mot, avec « … » si coupé. */ -}} {{- $r := .Params.resume -}} {{- if not $r -}} {{- range findRE `(?s)

.*?

` .Content -}} {{- /* sans le texte masqué « (nouvel onglet) » des liens externes */ -}} {{- $t := trim (. | replaceRE `.*?` "" | plainify | htmlUnescape) " \n" -}} {{- if and (not $r) $t }}{{ $r = $t | replaceRE `\s+` " " }}{{ end -}} {{- end -}} {{- end -}} {{- return (truncate 160 "…" $r) -}}
