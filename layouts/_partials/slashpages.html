{{- /* Redirections des slash pages (data/slashpages.yaml) : pour chaque article publié dont le dossier porte un nom de la liste, écrit //index.html (et ses équivalents) qui redirige vers l'article. Appelé une fois, par l'accueil. */ -}} {{- range where site.RegularPages "Section" "articles" -}} {{- $article := . -}} {{- $nom := .File.ContentBaseName -}} {{- range hugo.Data.slashpages -}} {{- if in . $nom -}} {{- range . -}} {{- $html := printf `%s%s` ($article.Title | htmlEscape) $article.Permalink $article.RelPermalink $article.RelPermalink ($article.Title | htmlEscape) -}} {{- (resources.FromString (printf "%s/index.html" .) $html).Publish -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
